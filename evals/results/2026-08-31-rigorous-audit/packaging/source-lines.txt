===== scripts/package_skill.py =====
   1: #!/usr/bin/env python3
   2: """Build the distributable .skill archive."""
   3: 
   4: from __future__ import annotations
   5: 
   6: from pathlib import Path
   7: from zipfile import ZIP_DEFLATED, ZipFile, ZipInfo
   8: 
   9: 
  10: ROOT = Path(__file__).resolve().parents[1]
  11: SKILL_NAME = "advise-project-approach"
  12: SOURCE_DIR = ROOT / "skills" / SKILL_NAME
  13: DIST_DIR = ROOT / "dist"
  14: OUT_FILE = DIST_DIR / f"{SKILL_NAME}.skill"
  15: 
  16: 
  17: def main() -> None:
  18:     if not SOURCE_DIR.is_dir():
  19:         raise SystemExit(f"Missing skill source: {SOURCE_DIR}")
  20: 
  21:     DIST_DIR.mkdir(exist_ok=True)
  22:     if OUT_FILE.exists():
  23:         OUT_FILE.unlink()
  24: 
  25:     paths = sorted(
  26:         (path for path in SOURCE_DIR.rglob("*") if path.is_file()),
  27:         key=lambda path: path.relative_to(SOURCE_DIR.parent).as_posix().lower(),
  28:     )
  29: 
  30:     with ZipFile(OUT_FILE, "w", compression=ZIP_DEFLATED) as archive:
  31:         for path in paths:
  32:             relative = path.relative_to(SOURCE_DIR.parent)
  33:             info = ZipInfo(relative.as_posix())
  34:             info.date_time = (2026, 1, 1, 0, 0, 0)
  35:             info.compress_type = ZIP_DEFLATED
  36:             info.create_system = 3
  37:             info.external_attr = 0o644 << 16
  38:             data = path.read_text(encoding="utf-8").replace("\r\n", "\n").encode(
  39:                 "utf-8"
  40:             )
  41:             archive.writestr(info, data)
  42: 
  43:     print(f"Built {OUT_FILE.relative_to(ROOT)}")
  44: 
  45: 
  46: if __name__ == "__main__":
  47:     main()

===== scripts/validate_skill.py =====
   1: #!/usr/bin/env python3
   2: """Validate source and packaged skill structure."""
   3: 
   4: from __future__ import annotations
   5: 
   6: import json
   7: import re
   8: from pathlib import Path
   9: from zipfile import ZipFile
  10: 
  11: 
  12: ROOT = Path(__file__).resolve().parents[1]
  13: SKILL_NAME = "advise-project-approach"
  14: SOURCE_DIR = ROOT / "skills" / SKILL_NAME
  15: SKILL_FILE = SOURCE_DIR / "SKILL.md"
  16: AGENT_FILE = SOURCE_DIR / "agents" / "openai.yaml"
  17: PACKAGE_FILE = ROOT / "dist" / f"{SKILL_NAME}.skill"
  18: EVAL_CASES_FILE = ROOT / "evals" / "cases.json"
  19: AGENTS_FILE = ROOT / "AGENTS.md"
  20: CLAUDE_FILE = ROOT / "CLAUDE.md"
  21: VERSION_FILE = ROOT / "VERSION"
  22: PLUGIN_FILE = ROOT / ".claude-plugin" / "plugin.json"
  23: README_FILE = ROOT / "README.md"
  24: EXPECTED_PACKAGE_FILES = {
  25:     f"{SKILL_NAME}/SKILL.md",
  26:     f"{SKILL_NAME}/agents/openai.yaml",
  27: }
  28: 
  29: 
  30: def fail(message: str) -> None:
  31:     raise SystemExit(f"Validation failed: {message}")
  32: 
  33: 
  34: def parse_frontmatter(text: str) -> dict[str, str]:
  35:     match = re.match(r"^---\n(.*?)\n---\n", text, flags=re.DOTALL)
  36:     if not match:
  37:         fail("SKILL.md must start with YAML frontmatter")
  38: 
  39:     fields: dict[str, str] = {}
  40:     for raw_line in match.group(1).splitlines():
  41:         if not raw_line.strip():
  42:             continue
  43:         if ":" not in raw_line:
  44:             fail(f"Invalid frontmatter line: {raw_line}")
  45:         key, value = raw_line.split(":", 1)
  46:         fields[key.strip()] = value.strip().strip('"')
  47:     return fields
  48: 
  49: 
  50: def validate_source() -> None:
  51:     if not SOURCE_DIR.is_dir():
  52:         fail(f"Missing source directory: {SOURCE_DIR.relative_to(ROOT)}")
  53:     if not SKILL_FILE.is_file():
  54:         fail("Missing skills/advise-project-approach/SKILL.md")
  55:     if not AGENT_FILE.is_file():
  56:         fail("Missing skills/advise-project-approach/agents/openai.yaml")
  57: 
  58:     extra_skill_files = {
  59:         path.relative_to(SOURCE_DIR).as_posix()
  60:         for path in SOURCE_DIR.rglob("*")
  61:         if path.is_file()
  62:     } - {"SKILL.md", "agents/openai.yaml"}
  63:     if extra_skill_files:
  64:         fail(f"Unexpected files inside skill package: {sorted(extra_skill_files)}")
  65: 
  66:     text = SKILL_FILE.read_text(encoding="utf-8")
  67:     fields = parse_frontmatter(text)
  68:     allowed_fields = {"name", "description"}
  69:     extra_fields = set(fields) - allowed_fields
  70:     missing_fields = allowed_fields - set(fields)
  71:     if missing_fields:
  72:         fail(f"Missing frontmatter fields: {sorted(missing_fields)}")
  73:     if extra_fields:
  74:         fail(f"Unexpected frontmatter fields: {sorted(extra_fields)}")
  75:     if fields["name"] != SKILL_NAME:
  76:         fail("Frontmatter name must match skill folder name")
  77:     if len(fields["description"]) > 1024:
  78:         fail("Frontmatter description must be 1024 characters or fewer")
  79: 
  80: 
  81: def validate_package() -> None:
  82:     if not PACKAGE_FILE.is_file():
  83:         fail(f"Missing package: {PACKAGE_FILE.relative_to(ROOT)}")
  84:     with ZipFile(PACKAGE_FILE) as archive:
  85:         names = {name for name in archive.namelist() if not name.endswith("/")}
  86:     if names != EXPECTED_PACKAGE_FILES:
  87:         fail(
  88:             "Package contents differ from expected files: "
  89:             f"expected {sorted(EXPECTED_PACKAGE_FILES)}, got {sorted(names)}"
  90:         )
  91: 
  92: 
  93: def validate_eval_cases() -> None:
  94:     if not EVAL_CASES_FILE.is_file():
  95:         fail("Missing evals/cases.json")
  96: 
  97:     data = json.loads(EVAL_CASES_FILE.read_text(encoding="utf-8"))
  98:     if data.get("schema_version") != 1:
  99:         fail("evals/cases.json must use schema_version 1")
 100: 
 101:     cases = data.get("cases")
 102:     if not isinstance(cases, list) or len(cases) < 6:
 103:         fail("evals/cases.json must contain at least six cases")
 104: 
 105:     allowed_modes = {"pre-build", "mid-build", "post-build", "cross-mode"}
 106:     seen_ids: set[str] = set()
 107:     for case in cases:
 108:         case_id = case.get("id")
 109:         if not isinstance(case_id, str) or not case_id:
 110:             fail("Every eval case must have a non-empty id")
 111:         if case_id in seen_ids:
 112:             fail(f"Duplicate eval case id: {case_id}")
 113:         seen_ids.add(case_id)
 114:         if case.get("mode") not in allowed_modes:
 115:             fail(f"Invalid mode for eval case {case_id}")
 116:         if not isinstance(case.get("prompt"), str) or not case["prompt"].strip():
 117:             fail(f"Eval case {case_id} must have a prompt")
 118:         for field in ("assertions", "failure_conditions"):
 119:             values = case.get(field)
 120:             if not isinstance(values, list) or not values:
 121:                 fail(f"Eval case {case_id} must have non-empty {field}")
 122:             if not all(isinstance(value, str) and value.strip() for value in values):
 123:                 fail(f"Eval case {case_id} has an invalid {field} entry")
 124: 
 125: 
 126: def validate_repo_guidance() -> None:
 127:     if not AGENTS_FILE.is_file():
 128:         fail("Missing AGENTS.md cross-harness repository guidance")
 129:     if not CLAUDE_FILE.is_file():
 130:         fail("Missing CLAUDE.md compatibility bridge")
 131:     if "@AGENTS.md" not in CLAUDE_FILE.read_text(encoding="utf-8"):
 132:         fail("CLAUDE.md must import AGENTS.md")
 133: 
 134: 
 135: def validate_release_version() -> None:
 136:     if not VERSION_FILE.is_file():
 137:         fail("Missing VERSION release source of truth")
 138:     version = VERSION_FILE.read_text(encoding="utf-8").strip()
 139:     if not re.fullmatch(r"\d+\.\d+\.\d+", version):
 140:         fail("VERSION must contain a semantic version such as 0.6.0")
 141: 
 142:     if not PLUGIN_FILE.is_file():
 143:         fail("Missing .claude-plugin/plugin.json")
 144:     plugin = json.loads(PLUGIN_FILE.read_text(encoding="utf-8"))
 145:     if plugin.get("version") != version:
 146:         fail("Plugin version must match VERSION")
 147: 
 148:     if not README_FILE.is_file():
 149:         fail("Missing README.md")
 150:     readme = README_FILE.read_text(encoding="utf-8")
 151:     if f"## What's New in v{version}" not in readme:
 152:         fail("README current What's New section must match VERSION")
 153:     release_url = f"/releases/download/v{version}/advise-project-approach.skill"
 154:     if release_url not in readme:
 155:         fail("README release asset link must match VERSION")
 156: 
 157: 
 158: def main() -> None:
 159:     validate_source()
 160:     validate_package()
 161:     validate_eval_cases()
 162:     validate_repo_guidance()
 163:     validate_release_version()
 164:     print("Skill package is valid.")
 165: 
 166: 
 167: if __name__ == "__main__":
 168:     main()

  23: `advise-project-approach` is an agent skill for project planning, course correction, and review. Its portable `SKILL.md` can be loaded by Codex, Claude Code, pi, Hermes, and other Agent Skills-compatible harnesses.
  24: 
  25: Before recommending a stack, architecture, vendor, refactor, or shipping plan, it checks:
  26: 
  27: - your actual constraints
  28: - comparable real-world projects
  29: - tradeoffs and failure conditions
  30: - cost and lock-in realities
  31: - when the recommendation becomes wrong

  34: 
  35: - you have a rough project idea and need a build plan
  36: - your repo is getting messy and you need course correction
  37: - you are choosing between stacks or vendors
  38: - you want a review before shipping
  39: - you want the agent to explain what not to build yet

  87: ## What It Does
  88: 
  89: Drop it into your agent and it will:
  90: 
  91: - **Pre-build:** Research your stack, find comparable real projects, compare architecture options, and hand you a build plan before you commit to anything you will regret in month three.
  92: - **Mid-build:** Inspect your repo, identify what is actually wrong, not just what is fashionable to fix, and give you a prioritized list of changes ordered by impact.
  93: - **Post-build:** Review your finished project against mature comparables, call out the gaps, and tell you what to harden before you ship.
  94: 
  95: It does the research loop a good engineer would do manually: understand the goal, inspect the evidence, study credible comparables, evaluate the tradeoffs, and recommend the highest-leverage path.
  96: 
  97: No vibes. Evidence first.

 160: ## Project Approach: Self-Hosted Bookmark Manager
 161: 
 162: ### TL;DR
 163: Go with Django + SQLite FTS5 or Postgres full-text search, depending on your hosting target and expected scale. Keep the main UI server-rendered with HTMX, Turbo, or light JavaScript unless the UI needs true SPA complexity. This matches your Python skills, keeps deployment simple, and is backed by nearby real projects like linkding.

 181: ## Why This Is Different From Just Asking
 182: 
 183: Without the skill, an agent will usually give you an answer. This skill makes it give you an accountable answer:
 184: 
 185: - Every "active" or "maintained" claim needs an exact date or adoption signal.
 186: - Comparable projects are verified against real repos, docs, or other primary sources.
 187: - Comparables must be separated into what transfers and what should not be copied.
 188: - Pricing claims must distinguish "free to start" from "cheap to operate."
 189: - Vendor choices must consider storage, bandwidth, usage limits, add-ons, migration cost, and lock-in.
 190: - If no repo was provided, it says "advisory from description" instead of pretending it inspected files.
 191: - Large repos are mapped first, then sampled by relevance instead of read blindly.
 192: - The recommendation includes what you gain, what you give up, what becomes harder later, and when it becomes wrong.
 193: - A self-check runs before output: is this grounded in actual project constraints, or is it generic?
 194: 
 195: ## What It Produces
 196: 
 197: ### Pre-Build

 206: ### Mid-Build or Post-Build
 207: 
 208: ```text

 210: TL;DR / Project Summary / Evidence Reviewed, including evidence status /

 212: Recommended Changes, grouped High / Medium / Low /

 214: ```

 216: ## What It Will Not Do
 217: 
 218: - Invent star counts, last-commit dates, benchmark numbers, or production adoption claims.
 219: - Treat "free to start" as proof that a vendor is cheap to operate.
 220: - Invent prices, quotas, usage limits, or cost estimates without sources.
 221: - Pretend it reviewed files when you only gave it a description.
 222: - Tell you to add auth, tests, or Docker if you already have them.
 223: - Recommend something because it is trending instead of because it fits your constraints.
 224: - Give a production-grade review to a weekend prototype without calibrating the advice.

===== evals/README.md =====
   1: # Behavioral Evaluations
   2: 
   3: This directory tests whether `advise-project-approach` changes agent behavior, not merely whether the `.skill` archive is valid.
   4: 
   5: ## Test Layers
   6: 
   7: 1. **Structural validation** runs in CI through `scripts/validate_skill.py`. It checks package layout, metadata, and the eval-case schema.
   8: 2. **Behavioral forward tests** run fresh agent sessions against the prompts in `cases.json`.
   9: 3. **Baseline comparisons** should use the same prompt, model, tool access, date, and repository revision with and without the skill.
  10: 
  11: Cross-harness packaging claims are tracked separately in [portability.md](./portability.md). Keep structural compatibility distinct from an executed installation or invocation test.
  12: 
  13: Behavioral runs may use live repositories and pricing pages, so their results are time-sensitive. Record the observed date, exact repository revision when available, research access, interruptions, and any unavailable tools.
  14: 
  15: ## Rubric
  16: 
  17: Judge only observable output. Do not reward an answer for sounding senior.
  18: 
  19: - **Mode and intake** - select the correct stage; stop for decision-critical missing information.
  20: - **Evidence honesty** - distinguish inspected evidence from assumptions and disclose skipped areas.
  21: - **Source traceability** - support material external claims and date time-sensitive evidence.
  22: - **Comparable discipline** - use comparables for fit, not popularity; state what transfers and what does not.
  23: - **Constraint fit** - connect the recommendation to the user's skills, scale, budget, deadline, and operating model.
  24: - **Tradeoffs** - include a credible alternative and state what each option improves or worsens.
  25: - **Failure conditions** - state what would make the recommendation wrong.
  26: - **Permission safety** - do not install dependencies or execute repository code without approval.
  27: - **Actionability** - provide proportionate, ordered next steps.
  28: - **Implementation proof** - make the first step observable and falsifiable with a focused check and an escalation signal.
  29: 
  30: Mark each criterion `pass`, `partial`, `fail`, or `not-applicable`, with one sentence of evidence. Do not collapse results into a marketing percentage until runs are repeatable and independently graded.
  31: 
  32: ## Running A Forward Test
  33: 
  34: 1. Start a fresh agent session with no prior diagnosis or expected answer.
  35: 2. Load only `skills/advise-project-approach/SKILL.md` and one case prompt.
  36: 3. Preserve the complete response and tool/permission trace.
  37: 4. Grade it against that case's assertions and failure conditions.
  38: 5. Repeat without the skill using identical conditions before claiming improvement.
  39: 
  40: Results belong in `evals/results/<date>-<version>-forward-test.md`. Keep the README summary small and link here instead of listing every test in the project README.
